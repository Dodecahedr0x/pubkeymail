```typescript
// src/services/encryption/encryption-service.ts
import nacl from 'tweetnacl';
import { db } from '../../database/connection.js';
import {
  EncryptionKeyPair,
  EncryptedData,
  EncryptionMetadata,
  UserEncryptionKey,
} from '../../types/encryption.js';

export class EncryptionService {
  async generateKeyPair(): Promise<EncryptionKeyPair> {
    const keyPair = nacl.box.keyPair();
    return {
      publicKey: Buffer.from(keyPair.publicKey).toString('hex'),
      privateKey: Buffer.from(keyPair.secretKey).toString('hex'),
    };
  }

  async encryptEmail(
    plaintext: string,
    recipientPublicKey: string
  ): Promise<EncryptedData> {
    const ephemeralKeyPair = nacl.box.keyPair();
    const nonce = nacl.randomBytes(24);
    const recipientPubKeyBytes = Buffer.from(recipientPublicKey, 'hex');
    const messageBytes = new TextEncoder().encode(plaintext);

    const ciphertext = nacl.box(
      messageBytes,
      nonce,
      recipientPubKeyBytes,
      ephemeralKeyPair.secretKey
    );

    return {
      ciphertext: Buffer.from(ciphertext).toString('base64'),
      nonce: Buffer.from(nonce).toString('hex'),
      ephemeralPublicKey: Buffer.from(ephemeralKeyPair.publicKey).toString('hex'),
    };
  }

  async decryptEmail(
    encrypted: EncryptedData,
    recipientPrivateKey: string
  ): Promise<string> {
    const ciphertext = Buffer.from(encrypted.ciphertext, 'base64');
    const nonce = Buffer.from(encrypted.nonce, 'hex');
    const ephemeralPubKey = Buffer.from(encrypted.ephemeralPublicKey, 'hex');
    const privateKey = Buffer.from(recipientPrivateKey, 'hex');

    const decrypted = nacl.box.open(
      ciphertext,
      nonce,
      ephemeralPubKey,
      privateKey
    );

    if (!decrypted) {
      throw new Error('Decryption failed - invalid ciphertext or key');
    }

    return new TextDecoder().decode(decrypted);
  }

  async storeUserPublicKey(userId: number, publicKey: string): Promise<void> {
    await db.query(
      `INSERT INTO user_encryption_keys (user_id, public_key, is_active)
       VALUES ($1, $2, true)
       ON CONFLICT (user_id) 
       DO UPDATE SET public_key = $2, is_active = true, updated_at = NOW()`,
      [userId, publicKey]
    );
  }

  async getUserPublicKey(userId: number): Promise<string | null> {
    const result = await db.query<{ public_key: string }>(
      `SELECT public_key FROM user_encryption_keys WHERE user_id = $1 AND is_active = true`,
      [userId]
    );
    return result.rows[0]?.public_key || null;
  }

  async getPublicKeyByAddress(address: string): Promise<string | null> {
    const result = await db.query<{ public_key: string }>(
      `SELECT uek.public_key 
       FROM user_encryption_keys uek
       JOIN users u ON uek.user_id = u.id
       JOIN blockchain_addresses ba ON u.primary_address_id = ba.id
       WHERE ba.address = $1 AND uek.is_active = true`,
      [address]
    );
    return result.rows[0]?.public_key || null;
  }
}

export const encryptionService = new EncryptionService();
```

```typescript
// src/api/routes/encryption-routes.ts
import { Router, Request, Response, NextFunction } from 'express';
import { z } from 'zod';
import { encryptionService } from '../../services/encryption/encryption-service.js';
import { authMiddleware } from '../middleware/auth-middleware.js';

const router = Router();

const registerKeySchema = z.object({
  publicKey: z.string().length(64, 'Public key must be 64 hex characters'),
});

router.post(
  '/keys',
  authMiddleware,
  async (req: Request, res: Response, next: NextFunction) => {
    try {
      const { publicKey } = registerKeySchema.parse(req.body);
      const userId = (req as any).user.userId;

      await encryptionService.storeUserPublicKey(userId, publicKey);

      res.json({
        success: true,
        message: 'Encryption key registered',
      });
    } catch (error) {
      next(error);
    }
  }
);

router.get(
  '/keys/:address',
  async (req: Request, res: Response, next: NextFunction) => {
    try {
      const { address } = req.params;
      const publicKey = await encryptionService.getPublicKeyByAddress(address);

      if (!publicKey) {
        res.status(404).json({
          error: { code: 'KEY_NOT_FOUND', message: 'No encryption key found for address' },
        });
        return;
      }

      res.json({ publicKey, encryptionSupported: true });
    } catch (error) {
      next(error);
    }
  }
);

export default router;
```

```typescript
// src/database/pool-monitor.ts
import { Pool, PoolClient } from 'pg';

export interface PoolStats {
  totalConnections: number;
  idleConnections: number;
  waitingClients: number;
  maxConnections: number;
  utilizationPercent: number;
}

export class PoolMonitor {
  private pool: Pool;
  private queryTimes: number[] = [];
  private readonly maxSamples = 100;

  constructor(pool: Pool) {
    this.pool = pool;
  }

  getStats(): PoolStats {
    const total = this.pool.totalCount;
    const idle = this.pool.idleCount;
    const waiting = this.pool.waitingCount;
    const max = (this.pool as any).options?.max || 20;

    return {
      totalConnections: total,
      idleConnections: idle,
      waitingClients: waiting,
      maxConnections: max,
      utilizationPercent: Math.round((total / max) * 100),
    };
  }

  recordQueryTime(ms: number): void {
    this.queryTimes.push(ms);
    if (this.queryTimes.length > this.maxSamples) {
      this.queryTimes.shift();
    }
  }

  getAverageQueryTime(): number {
    if (this.queryTimes.length === 0) return 0;
    const sum = this.queryTimes.reduce((a, b) => a + b, 0);
    return Math.round(sum / this.queryTimes.length);
  }

  isHealthy(): boolean {
    const stats = this.getStats();
    return stats.waitingClients < 5 && stats.utilizationPercent < 90;
  }
}
```

```typescript
import Bull, { Queue, Job } from 'bull';
import { config } from '../../config/index.js';

export interface EmailJobData {
  type: 'forward' | 'send' | 'cleanup';
  payload: Record<string, any>;
  priority?: number;
}

export interface JobResult {
  success: boolean;
  message?: string;
  error?: string;
}

class EmailQueue {
  private queue: Queue<EmailJobData> | null = null;
  private isInitialized = false;

  async initialize(): Promise<void> {
    if (this.isInitialized) return;

    this.queue = new Bull<EmailJobData>('email-processing', config.REDIS_URL, {
      defaultJobOptions: {
        removeOnComplete: 100,
        removeOnFail: 50,
        attempts: 3,
        backoff: {
          type: 'exponential',
          delay: 1000,
        },
      },
    });

    this.isInitialized = true;
  }

  async addJob(data: EmailJobData): Promise<Job<EmailJobData>> {
    if (!this.queue) {
      throw new Error('Queue not initialized');
    }

    return this.queue.add(data, {
      priority: data.priority || 5,
    });
  }

  async getStats(): Promise<{ waiting: number; active: number; completed: number; failed: number }> {
    if (!this.queue) {
      return { waiting: 0, active: 0, completed: 0, failed: 0 };
    }

    const [waiting, active, completed, failed] = await Promise.all([
      this.queue.getWaitingCount(),
      this.queue.getActiveCount(),
      this.queue.getCompletedCount(),
      this.queue.getFailedCount(),
    ]);

    return { waiting, active, completed, failed };
  }

  async close(): Promise<void> {
    if (this.queue) {
      await this.queue.close();
      this.queue = null;
      this.isInitialized = false;
    }
  }

  getQueue(): Queue<EmailJobData> | null {
    return this.queue;
  }
}

export const emailQueue = new EmailQueue();
```

```typescript
// src/services/cache/sns-cache.ts
import { getRedisClient } from './redis-client.js';
import { config } from '../../config/index.js';

const SNS_CACHE_PREFIX = 'sns:';
const DEFAULT_TTL = config.SNS_CACHE_TTL || 3600;

export interface CachedResolution {
  address: string;
  resolvedAt: number;
  expiresAt: number;
}

export class SNSCache {
  async get(name: string): Promise<CachedResolution | null> {
    const client = await getRedisClient();
    const key = `${SNS_CACHE_PREFIX}${name.toLowerCase()}`;
    const data = await client.get(key);

    if (!data) return null;

    try {
      return JSON.parse(data);
    } catch {
      return null;
    }
  }

  async set(name: string, address: string, ttl?: number): Promise<void> {
    const client = await getRedisClient();
    const key = `${SNS_CACHE_PREFIX}${name.toLowerCase()}`;
    const now = Date.now();

    const data: CachedResolution = {
      address,
      resolvedAt: now,
      expiresAt: now + (ttl || DEFAULT_TTL) * 1000,
    };

    await client.set(key, JSON.stringify(data), { EX: ttl || DEFAULT_TTL });
  }

  async invalidate(name: string): Promise<void> {
    const client = await getRedisClient();
    const key = `${SNS_CACHE_PREFIX}${name.toLowerCase()}`;
    await client.del(key);
  }

  async getStats(): Promise<{ hits: number; misses: number; size: number }> {
    const client = await getRedisClient();
    const keys = await client.keys(`${SNS_CACHE_PREFIX}*`);
    return { hits: 0, misses: 0, size: keys.length };
  }
}

export const snsCache = new SNSCache();
```

```typescript
// src/services/blockchain/ethereum-provider.ts
import { ethers, JsonRpcProvider } from 'ethers';
import {
  IBlockchainProvider,
  BlockchainType,
  AddressValidationResult,
  SignatureVerificationResult,
  NameResolutionResult,
  NameServiceType,
  NameResolutionError,
} from '../../types/blockchain.js';
import { config } from '../../config/index.js';

export class EthereumProvider implements IBlockchainProvider {
  private provider: JsonRpcProvider | null = null;

  getBlockchainType(): BlockchainType {
    return 'ethereum';
  }

  private getProvider(): JsonRpcProvider {
    if (!this.provider) {
      const rpcUrl = process.env.ETHEREUM_RPC_ENDPOINT || 'https://eth-mainnet.g.alchemy.com/v2/demo';
      this.provider = new JsonRpcProvider(rpcUrl);
    }
    return this.provider;
  }

  async validateAddress(address: string): Promise<AddressValidationResult> {
    try {
      if (!ethers.isAddress(address)) {
        return {
          valid: false,
          blockchain: 'ethereum',
          error: 'Invalid Ethereum address format',
        };
      }

      const checksumAddress = ethers.getAddress(address);
      return {
        valid: true,
        blockchain: 'ethereum',
        normalized: checksumAddress,
      };
    } catch (error) {
      return {
        valid: false,
        blockchain: 'ethereum',
        error: error instanceof Error ? error.message : 'Address validation failed',
      };
    }
  }

  async verifySignature(
    message: string,
    signature: string,
    address: string
  ): Promise<SignatureVerificationResult> {
    try {
      const recoveredAddress = ethers.verifyMessage(message, signature);
      const normalizedInput = ethers.getAddress(address);
      const normalizedRecovered = ethers.getAddress(recoveredAddress);

      const valid = normalizedInput.toLowerCase() === normalizedRecovered.toLowerCase();

      return {
        valid,
        address: normalizedInput,
        blockchain: 'ethereum',
        message,
        signature,
        error: valid ? undefined : 'Signature does not match address',
      };
    } catch (error) {
      return {
        valid: false,
        address,
        blockchain: 'ethereum',
        message,
        signature,
        error: error instanceof Error ? error.message : 'Signature verification failed',
      };
    }
  }

  async resolveNameService(
    name: string,
    nameService: NameServiceType
  ): Promise<NameResolutionResult> {
    if (nameService !== 'ENS') {
      throw new NameResolutionError(`Name service ${nameService} not supported for Ethereum`, 'ethereum');
    }

    try {
      const provider = this.getProvider();
      const address = await provider.resolveName(name);

      if (!address) {
        throw new NameResolutionError(`Could not resolve ENS name: ${name}`, 'ethereum');
      }

      return {
        name,
        nameService: 'ENS',
        blockchain: 'ethereum',
        resolvedAddress: address,
        resolvedAt: new Date(),
        cached: false,
      };
    } catch (error) {
      if (error instanceof NameResolutionError) throw error;
      throw new NameResolutionError(
        error instanceof Error ? error.message : 'ENS resolution failed',
        'ethereum'
      );
    }
  }

  async getAddressFromPublicKey(publicKey: string): Promise<string> {
    const address = ethers.computeAddress(publicKey);
    return address;
  }
}
```

```tsx
// web/providers/ToastProvider.tsx
'use client';

import { createContext, useContext, useState, useCallback, ReactNode } from 'react';
import { Toast, ToastType } from '../components/Toast';
import styles from '../components/Toast.module.css';

interface ToastItem {
  id: string;
  message: string;
  type: ToastType;
}

interface ToastContextType {
  showToast: (message: string, type?: ToastType) => void;
}

const ToastContext = createContext<ToastContextType | undefined>(undefined);

export function ToastProvider({ children }: { children: ReactNode }) {
  const [toasts, setToasts] = useState<ToastItem[]>([]);

  const showToast = useCallback((message: string, type: ToastType = 'info') => {
    const id = crypto.randomUUID();
    setToasts(prev => [...prev, { id, message, type }]);
  }, []);

  const removeToast = useCallback((id: string) => {
    setToasts(prev => prev.filter(t => t.id !== id));
  }, []);

  return (
    <ToastContext.Provider value={{ showToast }}>
      {children}
      <div className={styles.container}>
        {toasts.map(toast => (
          <Toast key={toast.id} {...toast} onClose={removeToast} />
        ))}
      </div>
    </ToastContext.Provider>
  );
}

export function useToast() {
  const context = useContext(ToastContext);
  if (!context) throw new Error('useToast must be used within ToastProvider');
  return context;
}
```

```tsx
// web/app/settings/forwarding/page.tsx
'use client';

import { useState, useEffect } from 'react';
import { useAuth } from '@/providers';
import { apiClient, ForwardingRule } from '@/lib/api-client';
import styles from './page.module.css';

export default function ForwardingPage() {
  const { token, isAuthenticated } = useAuth();
  const [rules, setRules] = useState<ForwardingRule[]>([]);
  const [loading, setLoading] = useState(true);
  const [newEmail, setNewEmail] = useState('');

  useEffect(() => {
    if (isAuthenticated && token) {
      loadRules();
    }
  }, [isAuthenticated, token]);

  const loadRules = async () => {
    try {
      const data = await apiClient.getForwardingRules(token!);
      setRules(data);
    } catch (error) {
      console.error('Failed to load rules:', error);
    } finally {
      setLoading(false);
    }
  };

  const createRule = async () => {
    if (!newEmail) return;
    try {
      await apiClient.createForwardingRule(token!, { destinationEmail: newEmail });
      setNewEmail('');
      loadRules();
    } catch (error) {
      console.error('Failed to create rule:', error);
    }
  };

  const toggleRule = async (ruleId: number, enabled: boolean) => {
    try {
      await apiClient.updateForwardingRule(token!, ruleId, { enabled });
      loadRules();
    } catch (error) {
      console.error('Failed to update rule:', error);
    }
  };

  const deleteRule = async (ruleId: number) => {
    if (!confirm('Delete this forwarding rule?')) return;
    try {
      await apiClient.deleteForwardingRule(token!, ruleId);
      loadRules();
    } catch (error) {
      console.error('Failed to delete rule:', error);
    }
  };

  if (loading) return <div className={styles.loading}>Loading...</div>;

  return (
    <div className={styles.container}>
      <h1>Email Forwarding</h1>
      
      <div className={styles.createForm}>
        <input
          type="email"
          placeholder="Forward emails to..."
          value={newEmail}
          onChange={(e) => setNewEmail(e.target.value)}
          className={styles.input}
        />
        <button onClick={createRule} className={styles.button}>
          Add Rule
        </button>
      </div>

      <div className={styles.rules}>
        {rules.length === 0 ? (
          <p className={styles.empty}>No forwarding rules configured</p>
        ) : (
          rules.map((rule) => (
            <div key={rule.id} className={styles.rule}>
              <div className={styles.ruleInfo}>
                <span className={styles.destination}>{rule.destinationEmail}</span>
                <span className={`${styles.status} ${rule.verified ? styles.verified : styles.pending}`}>
                  {rule.verified ? 'Verified' : 'Pending verification'}
                </span>
              </div>
              <div className={styles.ruleActions}>
                <label className={styles.toggle}>
                  <input
                    type="checkbox"
                    checked={rule.enabled}
                    onChange={(e) => toggleRule(rule.id, e.target.checked)}
                  />
                  <span className={styles.slider}></span>
                </label>
                <button 
                  onClick={() => deleteRule(rule.id)}
                  className={styles.deleteBtn}
                >
                  Delete
                </button>
              </div>
            </div>
          ))
        )}
      </div>
    </div>
  );
}
```
